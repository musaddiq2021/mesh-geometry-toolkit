cmake_minimum_required(VERSION 3.20)
project(MeshGeometryToolkit VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(MESH_SANITIZERS "" CACHE STRING "Optional GCC/Clang sanitizers, e.g. address,undefined")
option(MESH_STATIC_MINGW_RUNTIME "Link MinGW GCC/C++/POSIX runtimes statically for portable Windows executables" ON)

add_library(mesh_geometry STATIC src/geometry.cpp src/obj.cpp src/triangulation.cpp src/volume.cpp)
add_library(mesh::geometry ALIAS mesh_geometry)
target_include_directories(mesh_geometry PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(mesh_geometry PUBLIC cxx_std_20)
set_target_properties(mesh_geometry PROPERTIES CXX_EXTENSIONS OFF)

function(mesh_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /permissive- /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror
            $<$<CONFIG:Release>:-O2> $<$<CONFIG:Debug>:-D_GLIBCXX_ASSERTIONS>)
    endif()
    if(MESH_SANITIZERS)
        if(MSVC)
            message(FATAL_ERROR "MESH_SANITIZERS requires GCC or Clang with the requested sanitizer runtimes")
        endif()
        target_compile_options(${target} PRIVATE -fsanitize=${MESH_SANITIZERS} -fno-omit-frame-pointer)
        target_link_options(${target} PRIVATE -fsanitize=${MESH_SANITIZERS})
    endif()
endfunction()
mesh_warnings(mesh_geometry)

function(mesh_runtime target)
    if(MINGW AND CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND MESH_STATIC_MINGW_RUNTIME)
        if(MESH_SANITIZERS)
            message(FATAL_ERROR "Set MESH_STATIC_MINGW_RUNTIME=OFF for MinGW sanitizer builds; fully static sanitizer linkage is unsupported")
        endif()
        # The two GCC flags alone can leave a dynamic libwinpthread dependency.
        # -static also selects the toolchain's static POSIX thread archive.
        # Windows system/UCRT DLLs remain normal operating-system dependencies.
        target_link_options(${target} PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
endfunction()

add_executable(mesh-tool src/main.cpp)
target_link_libraries(mesh-tool PRIVATE mesh::geometry)
mesh_warnings(mesh-tool)
mesh_runtime(mesh-tool)

include(CTest)

    target_link_libraries(mesh-tests PRIVATE mesh::geometry)
    mesh_warnings(mesh-tests)
    mesh_runtime(mesh-tests)
    add_test(NAME mesh-unit COMMAND mesh-tests)
    add_test(NAME mesh-cli-help COMMAND mesh-tool --help)
    add_test(NAME mesh-cli-version COMMAND mesh-tool --version)
    set_tests_properties(mesh-cli-version PROPERTIES PASS_REGULAR_EXPRESSION "1.0.0")
    add_test(NAME mesh-cli-inspect COMMAND mesh-tool inspect ${CMAKE_CURRENT_SOURCE_DIR}/examples/tetrahedron.obj)
    set_tests_properties(mesh-cli-inspect PROPERTIES PASS_REGULAR_EXPRESSION "Surface area:")
    add_test(NAME mesh-cli-invalid COMMAND mesh-tool inspect ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/invalid.obj)
    set_tests_properties(mesh-cli-invalid PROPERTIES WILL_FAIL TRUE)
    add_test(NAME mesh-cli-integration COMMAND ${CMAKE_COMMAND}
        -DTOOL=$<TARGET_FILE:mesh-tool> -DSOURCE=${CMAKE_CURRENT_SOURCE_DIR}
        -DSCRATCH=${CMAKE_CURRENT_BINARY_DIR}/cli-test -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli.cmake)
endif()
